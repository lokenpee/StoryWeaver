<script>
document.addEventListener('DOMContentLoaded', function() {
    const rawData          = document.getElementById('rawData');
    const optionsList      = document.getElementById('optionsList');
    const optionsContainer = document.getElementById('optionsContainer');
    const cardHeader       = document.getElementById('cardHeader');
    const btnSend          = document.getElementById('btnSend');
    const btnAppend        = document.getElementById('btnAppend');
    const STORAGE_KEY      = 'st_choice_card_mode';

    let currentMode = localStorage.getItem(STORAGE_KEY) || 'append';

    function updateBtnUI() {
        btnSend.className = currentMode === 'send' ? 'mode-btn active' : 'mode-btn';
        btnAppend.className = currentMode === 'append' ? 'mode-btn active' : 'mode-btn';
    }

    function setMode(mode) {
        currentMode = mode;
        localStorage.setItem(STORAGE_KEY, currentMode);
        updateBtnUI();
    }

    btnSend.addEventListener('click', function(e) {
        e.stopPropagation();
        setMode('send');
    });

    btnAppend.addEventListener('click', function(e) {
        e.stopPropagation();
        setMode('append');
    });

    function getParentWindow() {
        try {
            return window.parent && window.parent !== window ? window.parent : window;
        } catch (err) {
            return window;
        }
    }

    function getParentDocument() {
        try {
            return getParentWindow().document || document;
        } catch (err) {
            return document;
        }
    }

    function getTriggerSlash() {
        const parentWin = getParentWindow();
        return parentWin.triggerSlash || window.triggerSlash || null;
    }

    function getInput() {
        return getParentDocument().querySelector('#send_textarea');
    }

    function dispatchInput(input) {
        input.dispatchEvent(new Event('input', { bubbles: true }));
    }

    function parseOptionLine(line) {
        const trimmed = line.trim();
        const fullMatch = trimmed.match(/^([A-Z])[：:]\s*([^：:]+)[：:]\s*(.+)$/);
        if (fullMatch) {
            return {
                label: fullMatch[1],
                title: fullMatch[2].trim(),
                content: fullMatch[3].trim(),
                sendText: fullMatch[3].trim()
            };
        }

        const simpleMatch = trimmed.match(/^([A-Z])[：:]\s*(.+)$/);
        if (simpleMatch) {
            return {
                label: simpleMatch[1],
                title: '',
                content: simpleMatch[2].trim(),
                sendText: simpleMatch[2].trim()
            };
        }

        return {
            label: '',
            title: '',
            content: trimmed,
            sendText: trimmed
        };
    }

    function createTextSpan(className, text) {
        const span = document.createElement('span');
        span.className = className;
        span.textContent = text;
        return span;
    }

    function createOptionItem(option) {
        const optionDiv = document.createElement('li');
        optionDiv.className = 'option-item';
        optionDiv.dataset.action = option.sendText;

        const mainDiv = document.createElement('div');
        mainDiv.className = 'option-main';

        if (option.label) {
            mainDiv.appendChild(createTextSpan('option-label', option.label));
        }

        const contentDiv = document.createElement('div');
        contentDiv.className = 'option-content';

        if (option.title) {
            contentDiv.appendChild(createTextSpan('option-title', option.title + '：'));
        }

        contentDiv.appendChild(document.createTextNode(option.content));
        mainDiv.appendChild(contentDiv);

        const actionsDiv = document.createElement('div');
        actionsDiv.className = 'option-actions';

        const editBtn = document.createElement('button');
        editBtn.className = 'send-btn';
        editBtn.type = 'button';
        editBtn.title = '修改并发送';
        editBtn.dataset.action = 'edit-send';
        editBtn.textContent = '📝➜';
        actionsDiv.appendChild(editBtn);

        optionDiv.appendChild(mainDiv);
        optionDiv.appendChild(actionsDiv);

        return optionDiv;
    }

    function setupOptionList(dataEl, targetContainer) {
        const rawText = dataEl.textContent.trim();
        const lines = rawText.split(/\r?\n+/).filter(function(line) {
            return line.trim() !== '';
        });
        const fragment = document.createDocumentFragment();

        lines.forEach(function(line) {
            fragment.appendChild(createOptionItem(parseOptionLine(line)));
        });

        targetContainer.innerHTML = '';
        targetContainer.appendChild(fragment);
    }

    function useSetInputFallback(text, shouldSend) {
        const input = getInput();
        const trigger = getTriggerSlash();

        if (input) {
            input.value = text;
            dispatchInput(input);
        } else if (trigger) {
            trigger('/setinput ' + text);
        } else {
            copyToClipboard(text);
            showToast('无法自动写入，已尝试复制', 'warning');
            return false;
        }

        if (shouldSend) {
            setTimeout(function() {
                try {
                    const sendBtn = getParentDocument().querySelector('#send_but');
                    if (sendBtn) sendBtn.click();
                } catch (err) {}
            }, 100);
        }

        return true;
    }

    function sendDirectly(text) {
        const trigger = getTriggerSlash();
        if (trigger) {
            try {
                trigger('/send ' + text + ' | /trigger await=true');
                showToast('已发送消息', 'success');
                return;
            } catch (err) {
                console.error('发送失败:', err);
            }
        }

        if (useSetInputFallback(text, true)) {
            showToast('已发送消息', 'success');
        } else {
            showToast('发送失败，请手动复制', 'error');
        }
    }

    function appendToInput(text) {
        try {
            const input = getInput();
            if (input) {
                const existing = input.value;
                const separator = existing.trim() === '' ? '' : ' ';
                input.value = existing + separator + text;
                dispatchInput(input);
                showToast('已追加到输入框', 'success');
                return;
            }

            const trigger = getTriggerSlash();
            if (trigger) {
                trigger('/setinput ' + text);
                showToast('无法读取输入框，已改为写入', 'warning');
                return;
            }

            copyToClipboard(text);
            showToast('无法自动追加，已尝试复制', 'warning');
        } catch (err) {
            console.error('追加失败:', err);
            showToast('追加失败，请手动复制', 'error');
        }
    }

    function handleOptionClick(text) {
        if (currentMode === 'send') {
            sendDirectly(text);
        } else {
            appendToInput(text);
        }
    }

    function setupEventDelegation() {
        optionsList.addEventListener('click', function(e) {
            const editBtn = e.target.closest('.send-btn');
            if (editBtn) {
                e.preventDefault();
                e.stopPropagation();
                const item = editBtn.closest('.option-item');
                if (item && item.dataset.action) {
                    editAndSend(item.dataset.action);
                }
                return;
            }

            const item = e.target.closest('.option-item');
            if (item && item.dataset.action) {
                handleOptionClick(item.dataset.action);
            }
        });
    }

    function setupCollapsible() {
        cardHeader.addEventListener('click', function(e) {
            if (e.target.closest('.mode-btn')) return;
            optionsContainer.classList.toggle('collapsed');
        });
    }

    function showCustomPrompt(title, value) {
        return new Promise(function(resolve) {
            const overlay = document.createElement('div');
            overlay.className = 'custom-modal-overlay';
            overlay.tabIndex = -1;

            const modal = document.createElement('div');
            modal.className = 'custom-modal';

            const titleEl = document.createElement('div');
            titleEl.className = 'modal-title';
            titleEl.textContent = title;

            const input = document.createElement('textarea');
            input.className = 'modal-input';
            input.value = value || '';

            const buttonsDiv = document.createElement('div');
            buttonsDiv.className = 'modal-buttons';

            function closeModal(result) {
                document.body.classList.remove('modal-open');
                if (document.body.contains(overlay)) {
                    document.body.removeChild(overlay);
                }
                resolve(result);
            }

            const cancelBtn = document.createElement('button');
            cancelBtn.className = 'modal-btn';
            cancelBtn.type = 'button';
            cancelBtn.textContent = '取消';
            cancelBtn.onclick = function() { closeModal(null); };

            const confirmBtn = document.createElement('button');
            confirmBtn.className = 'modal-btn modal-btn-primary';
            confirmBtn.type = 'button';
            confirmBtn.textContent = '发送';
            confirmBtn.onclick = function() { closeModal(input.value); };

            input.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && (e.ctrlKey || e.metaKey)) {
                    confirmBtn.click();
                }
            });

            overlay.addEventListener('keydown', function(e) {
                if (e.key === 'Escape') {
                    cancelBtn.click();
                }
            });

            overlay.addEventListener('click', function(e) {
                if (e.target === overlay) {
                    cancelBtn.click();
                }
            });

            buttonsDiv.appendChild(cancelBtn);
            buttonsDiv.appendChild(confirmBtn);
            modal.appendChild(titleEl);
            modal.appendChild(input);
            modal.appendChild(buttonsDiv);
            overlay.appendChild(modal);
            document.body.classList.add('modal-open');
            document.body.appendChild(overlay);

            setTimeout(function() {
                input.focus();
                input.select();
            }, 80);
        });
    }

    async function editAndSend(originalText) {
        const editedText = await showCustomPrompt('修改选项后发送', originalText);
        if (editedText === null) return;

        const finalText = editedText.trim();
        if (!finalText) {
            showToast('未输入内容', 'warning');
            return;
        }

        sendDirectly(finalText);
    }

    function copyToClipboard(text) {
        if (navigator.clipboard && window.isSecureContext) {
            navigator.clipboard.writeText(text).catch(function(err) {
                console.error('复制失败:', err);
                fallbackCopyTextToClipboard(text);
            });
        } else {
            fallbackCopyTextToClipboard(text);
        }
    }

    function fallbackCopyTextToClipboard(text) {
        const textArea = document.createElement('textarea');
        textArea.value = text;
        textArea.style.position = 'fixed';
        textArea.style.left = '-999999px';
        textArea.style.top = '-999999px';
        document.body.appendChild(textArea);
        textArea.focus();
        textArea.select();

        try {
            document.execCommand('copy');
        } catch (err) {
            console.error('降级复制失败:', err);
        }

        document.body.removeChild(textArea);
    }

    function showToast(message, type) {
        const toast = document.createElement('div');
        const bgColor = type === 'success' ? '#28a745' :
                        type === 'error' ? '#dc3545' : '#ffa500';

        toast.style.cssText = [
            'position: fixed',
            'top: 20px',
            'right: 20px',
            'background: ' + bgColor,
            'color: white',
            'padding: 12px 20px',
            'border-radius: 8px',
            'font-size: 14px',
            'z-index: 10000',
            'box-shadow: 0 4px 20px rgba(0, 0, 0, 0.2)',
            'animation: slideInRight 0.3s ease-out',
            'display: flex',
            'align-items: center',
            'gap: 8px',
            'max-width: calc(100vw - 40px)'
        ].join(';');
        toast.textContent = message;

        document.body.appendChild(toast);

        setTimeout(function() {
            toast.style.animation = 'slideOutRight 0.3s ease-in';
            setTimeout(function() {
                if (document.body.contains(toast)) {
                    document.body.removeChild(toast);
                }
            }, 300);
        }, 1800);
    }

    updateBtnUI();
    setupOptionList(rawData, optionsList);
    setupEventDelegation();
    setupCollapsible();
});
</script>
</body>
